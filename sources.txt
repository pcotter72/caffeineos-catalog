# sources.txt - what the CaffeineOS catalog offers.
#
# One [Program:Name] section per program. The scheduled GitHub Action runs
# tools/build_catalog.py, which turns this file into catalog.txt.
#
#   GitHub=owner/repo      newest release of that repository (not prereleases)
#   Asset=pattern          which release file to take (shell pattern, e.g. *.lha)
#   Aminet=dir/name        an Aminet package, e.g. util/dtype/JFIFdt44 (its .readme
#                          Version: is checked daily, the .lha fetched only on change)
#   URL=https://...        a fixed download (our own tools)
#   Archive=zip|lha|file   archive type; "file" = the download is the program itself
#   File=src > dest        a file inside the archive and where it is installed (repeatable);
#                          src may use * and ? (e.g. */Classes/...) but must match one file.
#                          Only the listed files are installed: no sources, docs or icons
#   Check=dest             the installed file whose $VER gives the installed version
#                          (must be the dest of one File= line)
#   Category=, Description= shown on the Amiga
#   Readme=URL             readme shown by CaffeineOSDT's "Show README" (optional)
#   ReadmeFile=path        or: a readme inside the archive, published under readmes/
#                          (Aminet entries get the package's .readme automatically)
#   AllowScript=yes        only for entries that are meant to replace a script
#                          (s bit); CaffeineOSDT 5.3.1+ refuses that otherwise,
#                          because scripts are often CaffeineOS's own
#
# Plain ASCII, LF line endings.
